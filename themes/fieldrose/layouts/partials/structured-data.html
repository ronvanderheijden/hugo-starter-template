{{/* Structured data for search engines, derived entirely from the site configuration and standard page fields — pages need no extra fields. Home page: Organization and WebSite. The logo is optional and only included when params.meta.logo is configured. Posts: Article with headline, dates, and author. */}} {{ if .IsHome }} {{ $org := dict "@context" "https://schema.org" "@type" "Organization" "name" site.Title "url" site.Home.Permalink }} {{ with site.Params.meta.logo }}{{ $org = merge $org (dict "logo" (. | absURL)) }}{{ end }} {{ $website := dict "@context" "https://schema.org" "@type" "WebSite" "name" site.Title "url" site.Home.Permalink }} {{ else if and .IsPage (eq .Section "posts") }} {{ $article := dict "@context" "https://schema.org" "@type" "Article" "headline" .Title "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "mainEntityOfPage" .Permalink "author" (dict "@type" "Organization" "name" (site.Params.meta.author | default site.Title)) "publisher" (dict "@type" "Organization" "name" site.Title) }} {{ with .Description }}{{ $article = merge $article (dict "description" .) }}{{ end }} {{ with .Params.image | default site.Params.meta.image }}{{ $article = merge $article (dict "image" (. | absURL)) }}{{ end }} {{ end }}
